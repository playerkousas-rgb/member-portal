/**
 * 訓練班報名收表 Script（每個訓練班 1 份）
 * ================================================================
 * 安裝步驟：
 *   1. 為呢個訓練班開一張新 Google Sheet。
 *   2. 擴充功能 → Apps Script → 將本檔案整份貼上 → 儲存。
 *   3. 執行 setupCourseSheet()（首次授權：Review permissions → Advanced → Allow），
 *      佢會自動建立「Regs」分頁並產生 API Key（只顯示一次，請即複製）。
 *   4. 部署 → 新增部署 → 網頁應用程式（執行身分：我自己；存取：所有人）→ 攞 /exec 網址。
 *   5. 將「課程名 + /exec 網址 + API Key」交俾主系統管理員，加入成員門戶後台
 *      （staff 後台 → 訓練班管理）。
 *
 * 之後所有報名會由主系統自動轉發嚟呢度，寫入本表嘅「Regs」分頁。
 * 負責人呢個訓練班嘅職員直接喺呢張 Sheet 睇名單、做 output、核對收費。
 */

var REG_SHEET = 'Regs';
var API_KEY_HASH = ''; // setupCourseSheet() 時自動填入（存 SHA-256）

// 入數紙存放資料夾（Google Drive 資料夾 ID）
// setupCourseSheet() 會自動喺你嘅 Drive 建立一個並記低；亦可喺選單「設定入數紙資料夾」改。
var RECEIPT_FOLDER_ID = '';

// 收表欄位（同主系統 submitCourseReg 送出嘅 payload 對應）
var REG_HEADERS = [
  'id', 'refCode', 'submittedAt', 'courseId', 'courseTitle',
  'nameZh', 'nameEn', 'gender', 'dob', 'phone', 'email',
  'memberType', 'section', 'badgeCode',
  'scoutDistrict', 'region', 'troop', 'scoutId', 'scoutPosition',
  'guardianConsent', 'guardianName', 'guardianRelation', 'guardianEmail', 'guardianPhone',
  'leaderConsent', 'leaderName', 'leaderPosition', 'leaderEmail',
  'payMethod', 'payerName', 'payAccount', 'receiptUrl', 'needReceipt', 'note',
];

// ===================== 一鍵建表 =====================

function setupCourseSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(REG_SHEET);
  if (!sh) {
    sh = ss.insertSheet(REG_SHEET);
    sh.getRange(1, 1, 1, REG_HEADERS.length).setValues([REG_HEADERS]);
  }
  var key = 'ck_' + Date.now().toString(36) + Math.random().toString(36).slice(2, 12);
  API_KEY_HASH = sha256_(key);
  // 將 API_KEY_HASH 寫入 Script Properties（持久儲存）
  PropertiesService.getScriptProperties().setProperty('API_KEY_HASH', API_KEY_HASH);

  // 建立／取回入數紙資料夾
  var folder = ensureReceiptFolder_();

  SpreadsheetApp.getUi().alert(
    '✅ 訓練班收表表已建立',
    '🔑 API Key（只顯示一次，請即複製）：\n────────────────\n' + key + '\n────────────────\n\n'
    + '📁 入數紙資料夾：\n' + (folder ? folder.getUrl() : '（未能建立，請手動設定）') + '\n\n'
    + '1. 部署 → 網頁應用程式（執行身分：我自己；存取：所有人）。\n'
    + '2. 將「課程名 + /exec 網址 + API Key」交俾主系統管理員加入成員門戶。\n'
    + '3. 報名時上傳嘅入數紙會自動存入上述資料夾。'
  );
}

// ===================== HTTP 入口 =====================

function doPost(e) {
  var body = {};
  try { body = JSON.parse(e.postData.contents); } catch (x) {}
  var action = (body.action || '').toString();
  if (action === 'addReg') return json(addReg_(body));
  return json(err('未知的 action: ' + action));
}

function doGet(e) {
  // 俾主系統／負責人查詢：?action=stats&apiKey=...
  var p = e.parameter;
  if (!authKey_(p.apiKey)) return json(err('Unauthorized: invalid or missing apiKey'));
  if ((p.action || '') === 'stats') return json(ok({ count: countRegs_() }));
  return json(err('未知的 action'));
}

// ===================== 收表 =====================

function addReg_(b) {
  if (!authKey_(b.apiKey)) return err('Unauthorized: invalid or missing apiKey');
  if (!b.nameZh || !b.phone || !b.email) return err('資料不完整');
  if (!b.receiptDataUrl) return err('請上傳入數紙截圖。未繳費將不獲處理申請');
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(REG_SHEET);
  if (!sh) return err('尚未執行 setupCourseSheet()');
  var id = 'cr_' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
  var ref = 'CRS-' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd') + '-' + Math.floor(Math.random() * 9000 + 1000);

  // 入數紙上傳去訓練班嘅 Drive 資料夾
  var receiptUrl = saveReceipt_(b.receiptDataUrl, b.receiptFileName || '入數紙', b.receiptMimeType || 'image/jpeg');
  if (!receiptUrl) return err('入數紙儲存失敗（未設定入數紙資料夾）。請聯絡負責職員。');

  appendRowObj_(sh, {
    id: id, refCode: ref, submittedAt: new Date().toISOString(),
    courseId: b.courseId || '', courseTitle: b.courseTitle || '',
    nameZh: b.nameZh, nameEn: b.nameEn || '', gender: b.gender || '', dob: b.dob || '',
    phone: b.phone, email: b.email || '',
    memberType: b.memberType || '', section: b.section || '', badgeCode: b.badgeCode || '',
    scoutDistrict: b.scoutDistrict || '', region: b.region || '', troop: b.troop || '',
    scoutId: b.scoutId || '', scoutPosition: b.scoutPosition || '',
    guardianConsent: b.guardianConsent || '', guardianName: b.guardianName || '',
    guardianRelation: b.guardianRelation || '', guardianEmail: b.guardianEmail || '',
    guardianPhone: b.guardianPhone || '',
    leaderConsent: b.leaderConsent || '', leaderName: b.leaderName || '',
    leaderPosition: b.leaderPosition || '', leaderEmail: b.leaderEmail || '',
    payMethod: b.payMethod || 'FPS', payerName: b.payerName || '', payAccount: b.payAccount || '',
    receiptUrl: receiptUrl, needReceipt: b.needReceipt || '', note: b.note || '',
  });
  return { refCode: ref };
}

function countRegs_() {
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(REG_SHEET);
  if (!sh) return 0;
  var last = sh.getLastRow();
  return last > 1 ? last - 1 : 0;
}

// ===================== 入數紙（Drive） =====================

function getReceiptFolder_() {
  var id = RECEIPT_FOLDER_ID || PropertiesService.getScriptProperties().getProperty('RECEIPT_FOLDER_ID') || '';
  if (!id) return null;
  try { return DriveApp.getFolderById(id); } catch (e) { return null; }
}

function ensureReceiptFolder_() {
  var folder = getReceiptFolder_();
  if (folder) return folder;
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    folder = DriveApp.createFolder('入數紙 — ' + ss.getName());
    PropertiesService.getScriptProperties().setProperty('RECEIPT_FOLDER_ID', folder.getId());
    RECEIPT_FOLDER_ID = folder.getId();
    return folder;
  } catch (e) {
    return null;
  }
}

// 將 base64 dataURL 存去入數紙資料夾，回傳可檢視連結
function saveReceipt_(dataUrl, fileName, mimeType) {
  var folder = getReceiptFolder_();
  if (!folder) return '';
  try {
    var base64 = String(dataUrl).split(',')[1] || String(dataUrl);
    var bytes = Utilities.base64Decode(base64);
    var blob = Utilities.newBlob(bytes, mimeType, fileName);
    var file = folder.createFile(blob);
    try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
    return file.getUrl();
  } catch (e) {
    return '';
  }
}

// ===================== 工具 =====================

function authKey_(key) {
  var stored = API_KEY_HASH || PropertiesService.getScriptProperties().getProperty('API_KEY_HASH') || '';
  if (!stored) return false;
  return sha256_(String(key || '')) === stored;
}
function ok(data)  { return { ok: true, data: data }; }
function err(msg)  { return { ok: false, error: msg }; }
function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }

function sha256_(str) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, str, Utilities.Charset.UTF_8)
    .map(function (b) { var v = (b < 0 ? b + 256 : b).toString(16); return v.length === 1 ? '0' + v : v; }).join('');
}

function sheetHeaders_(sh) {
  return sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(function (h) { return String(h).trim(); });
}
function appendRowObj_(sh, obj) {
  var headers = sheetHeaders_(sh);
  var arr = headers.map(function (h) { return obj[h] !== undefined ? obj[h] : ''; });
  sh.appendRow(arr);
}

function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🎓 訓練班')
    .addItem('一鍵建表（setupCourseSheet）', 'setupCourseSheet')
    .addItem('📁 設定入數紙資料夾', 'setReceiptFolderMenu')
    .addToUi();
}

function setReceiptFolderMenu() {
  var ui = SpreadsheetApp.getUi();
  var cur = getReceiptFolder_();
  var promptText = '輸入入數紙資料夾嘅 Google Drive 資料夾 ID：';
  if (cur) promptText = '現時資料夾：' + cur.getName() + '\n' + cur.getUrl() + '\n\n輸入新資料夾 ID 取代（留空 = 自動建立）：';
  var resp = ui.prompt('📁 設定入數紙資料夾', promptText, ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;
  var id = resp.getResponseText().trim();
  if (!id) {
    RECEIPT_FOLDER_ID = '';
    PropertiesService.getScriptProperties().deleteProperty('RECEIPT_FOLDER_ID');
    var f = ensureReceiptFolder_();
    ui.alert('完成', f ? '已自動建立：\n' + f.getUrl() : '建立失敗，請手動輸入資料夾 ID。');
    return;
  }
  try {
    var folder = DriveApp.getFolderById(id);
    RECEIPT_FOLDER_ID = id;
    PropertiesService.getScriptProperties().setProperty('RECEIPT_FOLDER_ID', id);
    ui.alert('完成', '入數紙會存入：\n' + folder.getName() + '\n' + folder.getUrl());
  } catch (e) {
    ui.alert('錯誤', '搵唔到呢個資料夾，請確認 ID 正確。');
  }
}
